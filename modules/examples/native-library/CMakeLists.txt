cmake_minimum_required(VERSION 3.21)
project(qualia_library_fixture LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(qualia_math SHARED ../library-suite/math.c)
set_target_properties(qualia_math PROPERTIES WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(UNIX)
  target_link_libraries(qualia_math PRIVATE m)
endif()
add_executable(module ../library-suite/cpp.cpp)
target_include_directories(module PRIVATE ../../../app/assets/module_sdk/include)
target_link_libraries(module PRIVATE qualia_math)
if(APPLE)
  set_target_properties(qualia_math PROPERTIES INSTALL_NAME_DIR "@rpath" BUILD_WITH_INSTALL_NAME_DIR ON)
  set_target_properties(module PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(module PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS module qualia_math RUNTIME DESTINATION bin LIBRARY DESTINATION bin)
